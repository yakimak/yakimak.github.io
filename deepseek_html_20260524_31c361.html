🚀 HTTP-сервер поверх TCP (реализация на Python)

📌 Как использовать:
1. Сохраните приведённый ниже Python-код в файл simple_http_server.py
2. Убедитесь, что у вас установлен Python 3.6+
3. Запустите в терминале: python simple_http_server.py
4. Откройте браузер и перейдите по адресу http://localhost:8080/
5. Сервер отдаёт HTML-страницы из папки ./www (создаст автоматически)
✨ Поддерживает: GET, обработку ошибок 404/403, нормализацию путей, безопасность.

📄 Исходный код сервера (Python 3)

#!/usr/bin/env python3
import socket
import os
import sys

HOST = 'localhost'
PORT = 8080
WEBROOT = './www'

def handle_client(client_socket, client_address):
    try:
        request_data = client_socket.recv(4096).decode('utf-8', errors='ignore')
        if not request_data:
            return
        request_line = request_data.splitlines()[0] if request_data.splitlines() else ''
        parts = request_line.split()
        if len(parts) < 2:
            send_error(client_socket, 400, "Bad Request")
            return
        method, path = parts[0], parts[1]
        if method.upper() != 'GET':
            send_error(client_socket, 405, "Method Not Allowed")
            return
        if '..' in path or path.startswith('/..'):
            send_error(client_socket, 403, "Forbidden")
            return
        if path.endswith('/'):
            path += 'index.html'
        relative_path = path.lstrip('/')
        if relative_path == '':
            relative_path = 'index.html'
        full_path = os.path.join(WEBROOT, relative_path)
        full_path = os.path.normpath(full_path)
        if not full_path.startswith(os.path.abspath(WEBROOT)):
            send_error(client_socket, 403, "Forbidden")
            return
        if not os.path.exists(full_path) or not os.path.isfile(full_path):
            send_error(client_socket, 404, "Not Found")
            return
        with open(full_path, 'rb') as f:
            content = f.read()
        response_headers = (
            "HTTP/1.1 200 OK\r\n"
            "Content-Type: text/html; charset=utf-8\r\n"
            f"Content-Length: {len(content)}\r\n"
            "Connection: close\r\n"
            "\r\n"
        )
        client_socket.sendall(response_headers.encode('utf-8'))
        client_socket.sendall(content)
    except Exception:
        send_error(client_socket, 500, "Internal Server Error")
    finally:
        client_socket.close()

def send_error(client_socket, code, message):
    body = f"<html><body><h1>{code} {message}</h1><hr><small>Simple HTTP Server</small></body></html>"
    response = (
        f"HTTP/1.1 {code} {message}\r\n"
        "Content-Type: text/html; charset=utf-8\r\n"
        f"Content-Length: {len(body)}\r\n"
        "Connection: close\r\n"
        "\r\n"
        f"{body}"
    )
    client_socket.sendall(response.encode('utf-8'))

def start_server():
    server_socket = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
    server_socket.setsockopt(socket.SOL_SOCKET, socket.SO_REUSEADDR, 1)
    try:
        server_socket.bind((HOST, PORT))
    except Exception as e:
        print(f"Bind error: {e}")
        sys.exit(1)
    server_socket.listen(5)
    print(f"Server running at http://{HOST}:{PORT}")
    if not os.path.exists(WEBROOT):
        os.makedirs(WEBROOT)
    index_path = os.path.join(WEBROOT, 'index.html')
    if not os.path.exists(index_path):
        with open(index_path, 'w', encoding='utf-8') as f:
            f.write("<!DOCTYPE html><html><head><title>Test</title></head><body><h1>HTTP сервер работает!</h1></body></html>")
    try:
        while True:
            client_sock, addr = server_socket.accept()
            print(f"Connection from {addr}")
            handle_client(client_sock, addr)
    except KeyboardInterrupt:
        print("\nShutting down.")
    finally:
        server_socket.close()

if __name__ == '__main__':
    start_server()
